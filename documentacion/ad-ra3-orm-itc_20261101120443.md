**app.py**
```python
"""Interfaz gráfica sencilla para operar sobre objetos persistentes."""
import argparse
import tkinter as tk
from tkinter import ttk, messagebox
from orm import Base, conectar, listar, alta, resumen_sql

p = argparse.ArgumentParser()
p.add_argument('--base', default='datos/itc.sqlite')
args = p.parse_args()
from pathlib import Path
Path(args.base).parent.mkdir(parents=True, exist_ok=True)
engine = conectar(args.base)
Base.metadata.create_all(engine)
ventana = tk.Tk()
ventana.title('ITC · Clientes y proyectos · ORM RA3')
ventana.geometry('850x480')
ttk.Label(ventana, text='Clientes y proyectos ITC', font=('Segoe UI', 18)).pack(pady=15)
tabla = ttk.Treeview(ventana, columns=('id','nombre','email','proyectos'), show='headings')
for campo in ('id','nombre','email','proyectos'):
    tabla.heading(campo, text=campo.capitalize())
tabla.pack(fill='both', expand=True, padx=20)
formulario = ttk.Frame(ventana)
formulario.pack(pady=15)
nombre = ttk.Entry(formulario)
email = ttk.Entry(formulario)
ttk.Label(formulario,text='Empresa').grid(row=0,column=0)
nombre.grid(row=0,column=1)
ttk.Label(formulario,text='Email').grid(row=0,column=2)
email.grid(row=0,column=3)
def refrescar():
    tabla.delete(*tabla.get_children())
    for fila in listar(engine): tabla.insert('', 'end', values=fila)
def guardar():
    if not nombre.get().strip() or '@' not in email.get():
        return messagebox.showerror('Datos incompletos','Introduce empresa y correo.')
    try:
        alta(engine,nombre.get().strip(),email.get().strip())
        refrescar()
    except Exception as error: messagebox.showerror('Operación revertida',str(error))
ttk.Button(formulario,text='Crear cliente y proyecto',command=guardar).grid(row=0,column=4,padx=10)
ttk.Button(ventana,text='Consultar resumen SQL',command=lambda:messagebox.showinfo('Estados',str(resumen_sql(engine)))).pack(pady=5)
refrescar()
ventana.mainloop()
engine.dispose()
```
**orm.py**
```python
"""Mapeo de las tablas SQLite creadas en C++ en RA2."""
from pathlib import Path
from sqlalchemy import create_engine, ForeignKey, String, select, text
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship, Session

class Base(DeclarativeBase):
    pass

class Cliente(Base):
    __tablename__ = 'clientes'
    id: Mapped[int] = mapped_column(primary_key=True)
    nombre: Mapped[str] = mapped_column(String, nullable=False)
    email: Mapped[str] = mapped_column(String, unique=True, nullable=False)
    estado: Mapped[str] = mapped_column(String, nullable=False)
    proyectos: Mapped[list['Proyecto']] = relationship(back_populates='cliente')

class Proyecto(Base):
    __tablename__ = 'proyectos'
    id: Mapped[int] = mapped_column(primary_key=True)
    cliente_id: Mapped[int] = mapped_column(ForeignKey('clientes.id'))
    nombre: Mapped[str] = mapped_column(String)
    estado: Mapped[str] = mapped_column(String)
    cliente: Mapped[Cliente] = relationship(back_populates='proyectos')

def conectar(ruta):
    return create_engine('sqlite:///' + str(Path(ruta).resolve()))

def listar(engine):
    with Session(engine) as sesion:
        return [(c.id, c.nombre, c.email, len(c.proyectos)) for c in sesion.scalars(select(Cliente).order_by(Cliente.nombre))]

def alta(engine, nombre, email):
    with Session(engine) as sesion, sesion.begin():
        cliente = Cliente(nombre=nombre, email=email, estado='prospecto')
        cliente.proyectos.append(Proyecto(nombre='Estudio inicial', estado='pendiente'))
        sesion.add(cliente)

def resumen_sql(engine):
    with engine.connect() as conexion:
        return conexion.execute(text('SELECT estado, COUNT(*) FROM clientes GROUP BY estado')).all()
```
**test_orm.py**
```python
import tempfile
import unittest
from pathlib import Path
from sqlalchemy.orm import Session
from sqlalchemy.exc import IntegrityError
from orm import Base, Cliente, conectar, alta, listar, resumen_sql

class Pruebas(unittest.TestCase):
    def test_persistencia_y_rollback(self):
        with tempfile.TemporaryDirectory() as carpeta:
            engine=conectar(Path(carpeta)/'prueba.sqlite')
            Base.metadata.create_all(engine)
            alta(engine,'Empresa ficticia','demo@example.test')
            self.assertEqual(listar(engine)[0][3],1)
            with self.assertRaises(IntegrityError):
                alta(engine,'Duplicado','demo@example.test')
            self.assertEqual(len(listar(engine)),1)
            with Session(engine) as s, s.begin():
                s.get(Cliente,1).estado='activo'
            self.assertEqual(resumen_sql(engine),[('activo',1)])
            engine.dispose()
if __name__=='__main__': unittest.main()
```
